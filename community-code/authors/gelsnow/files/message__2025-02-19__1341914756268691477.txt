// ==UserScript==
// @name         AWBW 占领显示 (Pure Version) 2.5
// @namespace    https://awbw.amarriner.com/
// @version      2.5
// @description  对脚本AWBW capture limit的优化，并针对平均房地产，进行判断，红绿蓝分别为，round大于floor（即玩家房地产很大概率超过平均房地产数），等于，和其他
// @match        https://awbw.amarriner.com/game.php?games_id*
// @icon         https://awbw.amarriner.com/terrain/aw1/neutralcity.gif
// @author       GelSnow（https://discord.gg/GHDqN23Y）
// @license      MIT
// @grant        none
// @inject-into  page
// ==/UserScript==

(function () {
    'use strict';

    // 获取玩家数量
    let playerCount = document.querySelectorAll('.player-overview-container').length;

    // 获取所有可占领的城市、总部、基地、港口的 ID
    let cities = document.querySelectorAll(
        '.game-building>img:first-child[src*="city.gif"], ' +
        '.game-building>img:first-child[src*="hq.gif"], ' +
        '.game-building>img:first-child[src*="base.gif"], ' +
        '.game-building>img:first-child[src*="port.gif"], ' +
        '.game-building>img:first-child[src*="city"], ' +
        '.game-building>img:first-child[src*="hq_"], ' +
        '.game-building>img:first-child[src*="base_"], ' +
        '.game-building>img:first-child[src*="port_"]'
    );
    let cityIds = Array.from(cities, city => +city.parentNode.getAttribute('data-building-id'));
    let totalCityCount = cityIds.length; // 记录总的房地产数量

    // 获取通讯塔 ID
    let comTowers = document.querySelectorAll('.game-building>img:first-child[src*="comtower"]');
    let comTowerIDs = Array.from(comTowers, tower => +tower.parentNode.getAttribute('data-building-id'));

    // 获取占领上限
    let captureLimit = Number(document.querySelectorAll('#showinfo td')[22]?.innerText) || 0;

    let playerOverviews = document.querySelectorAll('.player-overview-container');
    let playerTables = {};

    // 计算均衡的房地产数量，包含颜色判定逻辑
    function equalincome() {
        if (playerCount > 0) {
            let floatValue = totalCityCount / playerCount;
            let flooredValue = Math.floor(floatValue);
            let roundedValue = Math.round(floatValue);

            if (floatValue === flooredValue) {
                return {
                    value: flooredValue,
                    color: undefined  // 保持默认颜色
                };
            } else {
                let color = 'blue';  // 默认蓝色
                if (roundedValue > flooredValue) {
                    color = 'red';  // 四舍五入结果大于向下取整
                } else if (roundedValue === flooredValue) {
                    color = 'green';  // 四舍五入等于向下取整
                }
                return {
                    value: flooredValue,
                    color: color
                };
            }
        } else {
            return {
                value: '∞',
                color: 'red'
            };
        }
    }

    // 构建初始玩家表格
    playerOverviews.forEach(playerO => {
        let playerID = playerO.id.split('player')[1];
        let overviewMain = playerO.querySelector('.player-overview-main');

        let statsDiv = document.createElement('div');
        statsDiv.classList.add('flex', 'row', 'hv-center');
        statsDiv.style.gap = '3px';
        statsDiv.style.alignItems = 'flex-end';

        // 显示塔的统计信息
        if (comTowerIDs.length > 0) {
            let towerImg = document.createElement('img');
            towerImg.src = 'https://awbw.amarriner.com/terrain/aw1/neutralcomtower.gif';
            let towerSpan = document.createElement('span');
            towerSpan.className = 'PTowers-SumTowers';
            towerSpan.textContent = `0/${comTowerIDs.length}`;
            statsDiv.appendChild(towerImg);
            statsDiv.appendChild(towerSpan);
        }

        // 显示房地产的统计信息
        let propertyImg = document.createElement('img');
        propertyImg.src = 'https://awbw.amarriner.com/terrain/aw1/neutralcity.gif';
        let propertySpan = document.createElement('span');
        propertySpan.className = 'PProperties-CaptureLimit';
        let captureText = captureLimit > 0 ? `/${captureLimit}` : '';
        propertySpan.textContent = `(${equalincome().value})0${captureText}`;
        if (equalincome().color !== undefined) {
            propertySpan.style.color = equalincome().color;
        }
        statsDiv.appendChild(propertyImg);
        statsDiv.appendChild(propertySpan);

        overviewMain.appendChild(statsDiv);
        playerTables[playerID] = statsDiv;
    });

    // 节流函数，防止频繁调用
    function throttle(func, delay) {
        let lastCall = 0;
        return function () {
            let now = Date.now();
            if (now - lastCall >= delay) {
                func.apply(this, arguments);
                lastCall = now;
            }
        };
    }

    // 更新数据函数：每次调用都重新遍历 buildingsInfo 获取最新数据
    function updateData() {
        let playerProperties = {};
        let playerTowers = {};

        // 遍历全局变量 buildingsInfo 实时统计数据
        for (let i in buildingsInfo) {
            for (let z in buildingsInfo[i]) {
                let building = buildingsInfo[i][z];
                let playerID = building.buildings_players_id;
                if (playerID) {
                    if (cityIds.includes(building.buildings_id)) {
                        playerProperties[playerID] = (playerProperties[playerID] || 0) + 1;
                    }
                    if (comTowerIDs.includes(building.buildings_id)) {
                        playerTowers[playerID] = (playerTowers[playerID] || 0) + 1;
                    }
                }
            }
        }

        // 更新每个玩家的显示
        for (let playerID in playerTables) {
            let statsDiv = playerTables[playerID];

            // 更新塔的数据（如果存在塔统计）
            if (comTowerIDs.length > 0) {
                statsDiv.querySelector('.PTowers-SumTowers').textContent = `${playerTowers[playerID] || 0}/${comTowerIDs.length}`;
            }

            // 更新房地产的数据，应用 equalincome() 的值和颜色
            let captureText = captureLimit > 0 ? `/${captureLimit}` : '';
            let equalIncomeData = equalincome();
            let propertyText = `(${equalIncomeData.value})${playerProperties[playerID] || 0}${captureText}`;
            let propertySpan = statsDiv.querySelector('.PProperties-CaptureLimit');
            propertySpan.textContent = propertyText;
            if (equalIncomeData.color !== undefined) {
                propertySpan.style.color = equalIncomeData.color;
            }
        }
    }

    // 使用节流的方式调用 updateData，每隔 10 毫秒最多更新一次
    const throttledUpdateData = throttle(updateData, 10);

    // 使用 MutationObserver 监听 DOM 变化以触发更新
    const propertyObserver = new MutationObserver(() => throttledUpdateData());
    propertyObserver.observe(document, { childList: true, subtree: true });
})();
