function main(){
  tools.mooo_link();
}


const tools = {
  /** Replace view game with view game on awbw.moo */
  mooo_link : () => {
    // INPUT
    const map_nme = document.querySelector("#game-header-table > div.bold.underline.game-header-header")?.textContent?.trim()
    const vw_gme_btn = document.querySelector("#replay-misc-controls > table > tbody > tr > td > table > tbody > tr > td:nth-child(2) > a")
    // PROCESS
    const query = map_nme // + " rating>1200"
    // OUTPUT
    if (vw_gme_btn) {vw_gme_btn.href = "http://awbw.mooo.com/searchReplays.php?q=" + query}
  }
}


main()
