const DISABLE_INPUT_TIME = 60*10; // In seconds

const div = document.createElement("div");

(function() {
    'use strict';
    update_div();
    setTimeout(() => div.remove(), DISABLE_INPUT_TIME * 1000);
})();

function update_div() {
    div.id = "input_blocker" ;
    div.style = `
      background: #0000001f;
      position: fixed;
      inset: 0;
    `;
    document.body.appendChild(div);
}
